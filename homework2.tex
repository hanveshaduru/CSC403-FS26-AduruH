\section*{Section 1.4}

\subsection*{Problem 1}
Given
\[
A=\{1,3,12,35\},\qquad B=\{3,7,12,20\}
\]
and $C$ is the set of prime numbers.

\sub{a} $A\cap B$

The elements common to both $A$ and $B$ are $3$ and $12$.
\[
\boxed{A\cap B=\{3,12\}}
\]

\sub{b} $(A\cup B)\setminus C$

First, $A\cup B=\{1,3,7,12,20,35\}$. The prime numbers in this set are $3$ and $7$. Removing them gives
\[
\boxed{(A\cup B)\setminus C=\{1,12,20,35\}}
\]

\sub{c} $A\cup(B\setminus C)$

Since $B\setminus C=\{12,20\}$, we get
\[
\boxed{A\cup(B\setminus C)=\{1,3,12,20,35\}}
\]

Remark: no two of the three resulting sets are disjoint (each pair shares the element $12$). However, the set in (a) and the set in (b) are each a subset of the set in (c).

\subsection*{Problem 8}

\sub{a} Show that $(A\setminus B)\cap C=(A\cap C)\setminus B$.

Let $x$ be any element.
\begin{align*}
x\in(A\setminus B)\cap C &\iff (x\in A\land x\notin B)\land x\in C\\
&\iff (x\in A\land x\in C)\land x\notin B\\
&\iff x\in(A\cap C)\setminus B.
\end{align*}
Therefore,
\[
\boxed{(A\setminus B)\cap C=(A\cap C)\setminus B}
\]

\sub{b} Show that $(A\cap B)\setminus B=\varnothing$.

If $x\in(A\cap B)\setminus B$, then $x\in B$ and $x\notin B$, which is impossible. Therefore,
\[
\boxed{(A\cap B)\setminus B=\varnothing}
\]

\sub{c} Show that $A\setminus(A\setminus B)=A\cap B$.

Let $x$ be any element.
\begin{align*}
x\in A\setminus(A\setminus B) &\iff x\in A\land x\notin(A\setminus B)\\
&\iff x\in A\land\neg(x\in A\land x\notin B)\\
&\iff x\in A\land(x\notin A\lor x\in B)\\
&\iff x\in A\land x\in B\\
&\iff x\in A\cap B.
\end{align*}
Therefore,
\[
\boxed{A\setminus(A\setminus B)=A\cap B}
\]

\subsection*{Problem 14}
Show that $A\Delta(B\Delta C)=(A\Delta B)\Delta C$.

Use a truth table (T means the element is in the set):
\[
\begin{array}{ccc|c|c}
A&B&C&A\Delta(B\Delta C)&(A\Delta B)\Delta C\\
\hline
F&F&F&F&F\\
F&F&T&T&T\\
F&T&F&T&T\\
F&T&T&F&F\\
T&F&F&T&T\\
T&F&T&F&F\\
T&T&F&F&F\\
T&T&T&T&T
\end{array}
\]
The two columns are identical for every combination of $A,B,C$. Therefore,
\[
\boxed{A\Delta(B\Delta C)=(A\Delta B)\Delta C}
\]

\section*{Section 1.5}

\subsection*{Problem 2}

\sub{a} Let $P$ = Mary will sell her house, $Q$ = Mary can get a good price, $R$ = Mary can find a nice apartment. ``Only if'' indicates an implication:
\[
\boxed{P\to(Q\land R)}
\]

\sub{b} Let $P$ = having a good credit history, $Q$ = having an adequate down payment, $R$ = getting a mortgage. Since $P\land Q$ is a necessary condition for $R$,
\[
\boxed{R\to(P\land Q)}
\]

\sub{c} Let $P$ = John will drop out of school, $Q$ = someone stops John. ``$P$ unless $Q$'' means ``if not $Q$, then $P$.'' Therefore,
\[
\boxed{\neg Q\to P}
\]

\sub{d} Let $P$ = $x$ is divisible by $4$, $Q$ = $x$ is divisible by $6$, $R$ = $x$ is prime. Therefore,
\[
\boxed{(P\lor Q)\to\neg R}
\]

\subsection*{Problem 8}

\sub{a} Show that $(P\to Q)\land(Q\to R)$ is equivalent to $(P\to R)\land[(P\leftrightarrow Q)\lor(R\leftrightarrow Q)]$.
\[
\begin{array}{ccc|c|c}
P&Q&R&\text{LHS}&\text{RHS}\\
\hline
T&T&T&T&T\\
T&T&F&F&F\\
T&F&T&F&F\\
T&F&F&F&F\\
F&T&T&T&T\\
F&T&F&F&F\\
F&F&T&T&T\\
F&F&F&T&T
\end{array}
\]
The LHS and RHS columns are identical. Therefore,
\[
\boxed{\begin{aligned}
&(P\to Q)\land(Q\to R)\\
&\quad\equiv (P\to R)\land[(P\leftrightarrow Q)\lor(R\leftrightarrow Q)]
\end{aligned}}
\]

\sub{b} Show that $(P\to Q)\lor(Q\to R)$ is a tautology.
\[
\begin{array}{ccc|c}
P&Q&R&(P\to Q)\lor(Q\to R)\\
\hline
T&T&T&T\\
T&T&F&T\\
T&F&T&T\\
T&F&F&T\\
F&T&T&T\\
F&T&F&T\\
F&F&T&T\\
F&F&F&T
\end{array}
\]
The final column is always true. Therefore,
\[
\boxed{(P\to Q)\lor(Q\to R)\text{ is a tautology}}
\]

\subsection*{Problem 12}
Determine which formulas are equivalent:
\begin{align*}
\text{(a)}&\quad P\to(Q\to R)\\
\text{(b)}&\quad Q\to(P\to R)\\
\text{(c)}&\quad(P\to Q)\land(P\to R)\\
\text{(d)}&\quad(P\land Q)\to R\\
\text{(e)}&\quad P\to(Q\land R)
\end{align*}

\sub{a}
\begin{align*}
P\to(Q\to R) &\equiv P\to(\neg Q\lor R)\\
&\equiv\neg P\lor\neg Q\lor R
\end{align*}

\sub{b}
\begin{align*}
Q\to(P\to R) &\equiv Q\to(\neg P\lor R)\\
&\equiv\neg Q\lor\neg P\lor R\\
&\equiv\neg P\lor\neg Q\lor R
\end{align*}
Thus $\boxed{\text{(a)}\equiv\text{(b)}}$.

\sub{c}
\begin{align*}
(P\to Q)\land(P\to R) &\equiv(\neg P\lor Q)\land(\neg P\lor R)\\
&\equiv\neg P\lor(Q\land R)\\
&\equiv P\to(Q\land R)
\end{align*}
Thus $\boxed{\text{(c)}\equiv\text{(e)}}$.

\sub{d}
\begin{align*}
(P\land Q)\to R &\equiv\neg(P\land Q)\lor R\\
&\equiv\neg P\lor\neg Q\lor R
\end{align*}
This is the same as (a) and (b). Therefore $\boxed{\text{(a)}\equiv\text{(b)}\equiv\text{(d)}}$.

\medskip
\textbf{Final answer for \#12:}
\[
\boxed{\text{(a)}\equiv\text{(b)}\equiv\text{(d)}\qquad\text{and}\qquad\text{(c)}\equiv\text{(e)}}
\]
